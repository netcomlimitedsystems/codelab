def calculate_bmi(weight, height):
    return weight / (height ** 2)

# Try it yourself!
user_weight = 70
user_height = 1.75
bmi = calculate_bmi(user_weight, user_height)
print(f"Your BMI is: {bmi:.1f}")

# Real-time feedback
if bmi < 18.5:
    print("Underweight")
elif bmi < 25:
    print("Normal weight")
else:
    print("Overweight")
